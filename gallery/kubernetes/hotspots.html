name: GitStats Report
on:
  push:
    branches: [main]

  contents: write

  gitstats:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7
        with:
          fetch-depth: 0
      - uses: shenxianpeng/gitstats-action@v1
        with:
